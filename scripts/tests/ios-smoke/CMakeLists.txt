cmake_minimum_required(VERSION 3.22)
project(IosSetupSmokeTests LANGUAGES CXX)
enable_testing()
add_executable(IosSetupSmokeTests Test.cpp)
target_compile_features(IosSetupSmokeTests PRIVATE cxx_std_20)
target_include_directories(IosSetupSmokeTests PRIVATE ../../../src/App/src)
if(MSVC)
    target_compile_options(IosSetupSmokeTests PRIVATE /W4 /WX)
else()
    target_compile_options(IosSetupSmokeTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME IosSetupSmokeTests COMMAND IosSetupSmokeTests)
